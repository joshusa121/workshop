from fractions import Fraction as F
import math, json

def fl10(x): return (math.floor(x) // 10) * 10          # 10원 미만 버림
def fl1(x):  return math.floor(x)                        # 1원 미만 버림

def prog(base, table):                                   # 누진세: [(상한, 세율)]
    base = max(0, base); tax = F(0); lo = 0
    for up, r in table:
        hi = base if up is None else min(base, up)
        if hi > lo: tax += (hi - lo) * r
        if up is None or base <= up: break
        lo = up
    return tax

ITA = [(14_000_000,F(6,100)),(50_000_000,F(15,100)),(88_000_000,F(24,100)),(150_000_000,F(35,100)),
       (300_000_000,F(38,100)),(500_000_000,F(40,100)),(1_000_000_000,F(42,100)),(None,F(45,100))]  # 소득세법 §55①
CTA = [(200_000_000,F(10,100)),(20_000_000_000,F(20,100)),(300_000_000_000,F(22,100)),(None,F(25,100))]  # 법인세법 §55①1호
NHIS = F(719,10000)        # 건보법 시행령 §44①
LTC  = F(9448,1000000)     # 노인장기요양보험법 시행령 §4 (100만분의 9,448)
NPS_FULL, NPS_HALF = F(95,1000), F(475,10000)   # 국민연금법 부칙(2025.4.2) 제4조 — 2026년
BK_INDIV, BK_CORP = 90_000*12 + 420_000, 125_000*12 + 900_000   # 전제(assumptions)

def nps_bases(annual):     # 월액 천원 미만 절사, 상반기 40만~637만 / 하반기 41만~659만
    m = (annual // 12) // 1000 * 1000

    d = prog(s, [(5_000_000,F(70,100)),(15_000_000,F(40,100)),(45_000_000,F(15,100)),(100_000_000,F(5,100)),(None,F(2,100))])
    return min(fl1(d), 20_000_000)

def earned_credit(calc, salary):   # §59①②
    c = calc*F(55,100) if calc <= 1_300_000 else 715_000 + (calc-1_300_000)*F(30,100)
    if salary <= 33_000_000: lim = 740_000
    elif salary <= 70_000_000: lim = max(660_000, 740_000 - (salary-33_000_000)*F(8,1000))
    elif salary <= 120_000_000: lim = max(500_000, 660_000 - (salary-70_000_000)*F(1,2))
    else: lim = max(200_000, 500_000 - (salary-120_000_000)*F(1,2))
    return fl1(min(c, lim))

def yu_cap(biz):           # 조특법 §86의3① 1~4호
    return 6_000_000 if biz<=40_000_000 else 5_000_000 if biz<=60_000_000 else 4_000_000 if biz<=100_000_000 else 2_000_000

def employee_ins(salary):  # 본인분(= 사용자분)
    nps = fl1(nps_bases(salary)*NPS_HALF); hi = fl1(salary*NHIS/2); ltc = fl1(salary*LTC/2)
    return nps, hi, ltc

def wage_tax(salary):      # 근로소득 과표·산출 (공제: 기본 150만 + 연금 + 건보·요양 본인분)
    nps, hi, ltc = employee_ins(salary)
    inc = salary - earned_ded(salary)
    base = max(0, inc - 1_500_000 - nps - hi - ltc)
    return inc, nps+hi+ltc, base

def extra_nhis(div):       # 건보법 §71①·영 §41④ 2천만 공제·규칙 §44① 1천만 절벽 (상한 미적용)
    if div <= 10_000_000: return 0
    m = F(max(0, div - 20_000_000), 12)
    return fl1(m*NHIS)*12 + fl1(m*LTC)*12

def owner(salary, div):    # 법인 대표: 급여 + 배당 (§14③6호·§62 비교과세·§56 배당세액공제)
    inc, ins, wbase = wage_tax(salary)
    wcalc = fl10(prog(wbase, ITA))
    if div <= 20_000_000:
        calc = wcalc + fl10(div*F(14,100)); dcredit = 0
    else:
        gross = fl1((div-20_000_000)*F(10,100))
        m1 = fl10(prog(wbase + (div-20_000_000) + gross, ITA)) + fl10(20_000_000*F(14,100))
        m2 = wcalc + fl10(div*F(14,100))
        dcredit = min(gross, max(0, m1-m2)); calc = max(m1, m2) - dcredit
    credits = earned_credit(wcalc, salary) + 130_000
    det = max(0, calc - credits); local = fl10(det*F(1,10))     # R1: 결정세액 × 10%
    nhx = extra_nhis(div)
    disp = salary + div - det - local - ins - nhx
    return dict(wcalc=wcalc, calc=calc, dividendCredit=dcredit, determined=det, local=local, ins=ins, nhx=nhx, disposable=disp)

def family(salary):
    nps, hi, ltc = employee_ins(salary); ins = nps+hi+ltc
    inc = salary - earned_ded(salary); base = max(0, inc - 1_500_000 - ins)
    calc = fl10(prog(base, ITA)); det = max(0, calc - earned_credit(calc, salary) - 130_000)
    local = fl10(det*F(1,10)); burden = det + local + ins          # R1: 결정세액 × 10%
    return dict(burden=burden, disposable=salary-burden, corpCost=salary+ins)

def individual(profit, sincere=False, relief=0, yu=True):
    b = nps_bases(profit); nps = fl1(b*NPS_FULL); hi = fl1(profit*NHIS); ltc = fl1(profit*LTC)
    ins = nps + hi + ltc
    health = hi + ltc                                           # R2: 지역 건강·장기요양 필요경비 (영 §55①11의3)
    business = max(0, profit - health)
    yud = yu_cap(business) if yu else 0
    base = max(0, business - 1_500_000 - nps - yud)
    calc = fl10(prog(base, ITA))
    rel = fl10(calc*F(relief).limit_denominator())
    cost = 1_500_000 if sincere else 0; scredit = min(fl1(cost*F(6,10)), 1_200_000)
    det = max(0, calc - 70_000 - rel - scredit)
    local = fl10(det*F(1,10))                                    # R1: 결정세액 × 10%
    disp = profit - det - local - ins - BK_INDIV - cost
    return dict(ins=ins, health=health, business=business, yu=yud, base=base, calc=calc, relief=rel, determined=det, local=local, disposable=disp)

def corporate(profit, salary, div, fam_n=0, fam_salary=36_000_000, sincere=False):
    nps, hi, ltc = employee_ins(salary); emp = nps+hi+ltc
    famc = family(fam_salary)['corpCost']*fam_n
    base = max(0, profit - salary - emp - famc)
    ctax = fl10(prog(base, CTA)); clocal = fl10(ctax*F(1,10))
    cost = 1_500_000 if sincere else 0; scredit = min(fl1(cost*F(6,10)), 1_500_000)
    cash = profit - salary - emp - famc - (ctax - scredit) - clocal - BK_CORP - cost
    d = min(div, max(0, cash)); o = owner(salary, d)
    return dict(taxBase=base, corpTax=ctax, corpLocal=clocal, cash=cash, dividend=d, retained=cash-d,
                owner=o, combined=o['disposable'] + cash - d)

out = {}
# ① 이익 1.5억 · 가족 1명(3,600만) · 배당 1천만 · 노란우산 ON · 급여 기본 7,200만
out['A1'] = dict(indiv=individual(150_000_000), corp=corporate(150_000_000, 72_000_000, 10_000_000, fam_n=1), family=family(36_000_000))
# ② 이익 3억 · 성실신고 대상(서비스 매출 8억) · 정관 3배 · 창업감면 50% 잔여 2년
c2 = corporate(300_000_000, 72_000_000, 20_000_000, sincere=True)
lim1 = 72_000_000*10//10; itax_lim = lim1*2; deduct_lim = lim1*3
inc = max(0, min(c2['retained'], deduct_lim) - itax_lim)
b0 = c2['taxBase']; b1 = max(0, b0 - inc)
t0 = fl10(prog(b0, CTA)); t1 = fl10(prog(b1, CTA))
c2['mult3'] = dict(deductibleIncrease=inc, corpTaxDecrease=(t0-t1) + (fl10(t0*F(1,10)) - fl10(t1*F(1,10))))
out['A2'] = dict(indiv=individual(300_000_000, sincere=True, relief=0.5), corp=c2)
# ③ 이익 6억 · 서울(과밀) · 부동산 5억 · 배당 3천만 · 자본금 5천만
reg = max(fl1(50_000_000*F(4,1000)), 112_500)*3; edu = fl1(reg*F(2,10))
acq = fl1(500_000_000*(F(4,100)*3 - F(2,100)*2)); acq_rel = acq - fl1(acq*F(1,2))
out['A3'] = dict(indiv=individual(600_000_000), corp=corporate(600_000_000, 72_000_000, 30_000_000),
                 costs=dict(registration=reg, education=edu, regTotal=reg+edu, acquisition=acq, acquisitionAfterRelief=acq_rel),
                 nhx=extra_nhis(30_000_000))
